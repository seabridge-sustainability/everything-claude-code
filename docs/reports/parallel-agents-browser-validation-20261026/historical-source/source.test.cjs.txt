'use strict';
const test=require('node:test'),assert=require('node:assert/strict'),fs=require('node:fs'),c=require('./common.cjs');
test('real navigation interval intersection only',()=>{const step=(name,a,b,navigation=true)=>({name,startNs:'0',endNs:'10000',navigationStartNs:String(a),navigationEndNs:String(b),navigation,passed:true});assert.deepEqual(c.overlap([step('goto',100,200)],[step('back',150,250)]),[{desktop:'goto',mobile:'back',startNs:'150',endNs:'200',durationNs:'50'}]);assert.deepEqual(c.overlap([step('goto',100,150)],[step('back',150,250)]),[]);assert.deepEqual(c.overlap([step('cookies',100,200,false)],[step('back',150,250)]),[]);assert.deepEqual(c.overlap([{...step('goto',100,200),passed:false}],[step('back',150,250)]),[]);});
test('unknown or shared process can never satisfy closure',()=>{assert.equal(c.stopped(null),false);assert.equal(c.stopped({pid:2101271,pgid:2101271,startTicks:'43180185'}),false);assert.equal(c.stopped(c.proc(process.pid)),false);});
test('fixtures have no external resources or script and links stay local',()=>{for(const n of ['index','details','form']){const html=fs.readFileSync(__dirname+'/fixtures/'+n+'.html','utf8');assert.doesNotMatch(html,/<script|https?:|src\s*=/i);for(const m of html.matchAll(/(?:href|action)="([^"]+)"/g))assert.ok(['details.html','index.html','#mobile-section'].includes(m[1]));}});
test('worker does not configure a port, server, shared CDP or measured delay',()=>{const src=fs.readFileSync(__dirname+'/worker.cjs','utf8');assert.doesNotMatch(src,/launchServer|connectOverCDP|remote-debugging-port=/);assert.doesNotMatch(src.slice(src.indexOf("diagnosticPhase='navigation';recorder.enable();")),/await c.wait|setTimeout|waitForTimeout/);assert.match(src,/context\.route/);assert.match(src,/route\.abort/);assert.match(src,/fresh synthetic profile required/);});
test('canonical heavy envelope and actual target admission are distinct',()=>{const src=fs.readFileSync(__dirname+'/controller.cjs','utf8');assert.match(src,/'heavy-check','local-machine'/);assert.match(src,/'browser-target'/);assert.doesNotMatch(src,/'native-client'/);assert.match(src,/q\.recheck\(claim\)\.admitted/);});
test('partial pre-start reconciliation keeps exact real tokens and releases heavy last',()=>{const i={coordinatorActor:'root',workers:[{role:'desktop',actor:'a'},{role:'mobile',actor:'b'}]},token='a'.repeat(48),heavy={id:'launch-envelope',owner:'root',state:'running',token},target={id:'desktop',owner:'a',state:'running',token:'b'.repeat(48)};assert.deepEqual(c.reconciliationClaims(i,[heavy,target,{id:'mobile',owner:'b',state:'waiting'}]),[{id:'desktop',owner:'a',token:'b'.repeat(48)},{id:'launch-envelope',owner:'root',token}]);assert.throws(()=>c.reconciliationClaims(i,[{...target,owner:'other'}]),/owned retained claim/);assert.throws(()=>c.reconciliationClaims(i,[{...target,token:null}]),/owned retained claim/);});
test('atomic create-only publication cannot replace consumed receipt',()=>{const scope=__dirname+'/publication-test';c.writable('astra-local-parallel-navigation-v2-source-01a175cd',[scope]);const dir=fs.mkdtempSync(scope+'-'),p=dir+'/READY.json';try{c.publishNew(p,{ready:true,payload:'x'.repeat(10000)});assert.equal(JSON.parse(fs.readFileSync(p)).payload.length,10000);assert.throws(()=>c.publishNew(p,{ready:false}),/EEXIST/);assert.equal(JSON.parse(fs.readFileSync(p)).ready,true);assert.deepEqual(fs.readdirSync(dir),['READY.json']);}finally{if(fs.existsSync(p))fs.unlinkSync(p);fs.rmdirSync(dir);}});
