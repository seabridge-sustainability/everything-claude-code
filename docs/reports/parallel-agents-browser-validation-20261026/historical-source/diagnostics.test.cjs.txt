'use strict';
const test=require('node:test'),assert=require('node:assert/strict'),crypto=require('node:crypto'),d=require('./diagnostics.cjs'),c=require('./common.cjs');
const base='file:///owned/fixtures/';
test('bounded child output remains diagnostic without result or structured stderr',()=>{const x=d.createCollector(base,32),a=Buffer.from('ReferenceError: missing fixture symbol\n'),b=Buffer.alloc(100,120);x.push('stderr',a);x.push('stderr',b);const r=x.finish();assert.equal(r.stderr.bytes,a.length+b.length);assert.equal(r.stderr.retainedBytes,32);assert.equal(r.stderr.truncated,true);assert.equal(r.stderr.text,a.subarray(0,32).toString());assert.equal(r.stderr.sha256,crypto.createHash('sha256').update(a).update(b).digest('hex'));assert.deepEqual(r.structured,[]);});
test('structured registry error retains exact phase/classification and strips foreign URLs',()=>{const x=d.createCollector(base),e=Error('coordination registry busy; retry once, then report stale lock (never steal it)');x.push('stderr',JSON.stringify(d.diagnostic(e,'result-publication',base))+'\n');x.push('stdout','https://private.invalid/secret file:///foreign/private file:///owned/fixtures/index.html');const r=x.finish();assert.equal(r.structured[0].classification,'REGISTRY_BUSY');assert.equal(r.structured[0].phase,'result-publication');assert.doesNotMatch(r.stdout.text,/private\.invalid|foreign\/private/);assert.match(r.stdout.text,/file:\/\/\/owned\/fixtures\/index\.html/);});
test('async event refusal never throws through emitter and is caught at step boundary',()=>{const events=[],r=d.eventRecorder({base,clock:()=>'1',events});r.enable();assert.doesNotThrow(()=>r.record('framenavigated','about:blank'));assert.equal(r.failure().classification,'FIXTURE_EVENT_REFUSED');assert.throws(()=>r.assertClean(),/fixture event refused/);assert.deepEqual(events,[]);assert.doesNotMatch(JSON.stringify(r.failure()),/about:blank/);});
test('event cap fails closed while disabled closure events are ignored',()=>{const events=[],r=d.eventRecorder({base,clock:()=>'2',events,limit:1});r.record('framenavigated','about:blank');assert.equal(r.failure(),null);r.enable();r.record('request',base+'index.html');assert.doesNotThrow(()=>r.assertClean());assert.equal(events.length,1);assert.doesNotThrow(()=>r.record('requestfinished',base+'index.html'));assert.throws(()=>r.assertClean(),/bounded navigation events/);});
test('registry busy gets one bounded retry only; unrelated failures never retry',()=>{const busy='coordination registry busy; retry once, then report stale lock (never steal it)',waits=[];let n=0;assert.equal(c.once(()=>{if(n++===0)throw Error(busy);return 'ok';},ms=>waits.push(ms)),'ok');assert.equal(n,2);assert.deepEqual(waits,[750]);n=0;assert.throws(()=>c.once(()=>{n++;throw Error(busy);},()=>{}),/registry busy/);assert.equal(n,2);n=0;assert.throws(()=>c.once(()=>{n++;throw Error('permission refused');},()=>{throw Error('must not wait');}),/permission refused/);assert.equal(n,1);});
test('UTF-8 redaction expansion stays within actual retained byte cap',()=>{const input='漢'.repeat(2000)+'data:a '.repeat(300),x=d.createCollector(base);x.push('stderr',input);const r=x.finish().stderr;assert.equal(r.bytes,8100);assert.equal(r.capturedRawBytes,8100);assert.equal(r.truncated,true);assert.ok(Buffer.byteLength(r.text)<=8192);assert.equal(r.retainedBytes,Buffer.byteLength(r.text));assert.doesNotMatch(r.text,/\uFFFD|data:a/);const y=d.createCollector(base,8);y.push('stderr','漢'.repeat(4));const t=y.finish().stderr;assert.ok(Buffer.byteLength(t.text)<=8);assert.equal(t.truncated,true);});
