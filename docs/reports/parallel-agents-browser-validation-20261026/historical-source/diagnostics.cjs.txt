'use strict';
// Diagnostics are exclusively for this source-pinned, fresh synthetic fixture.
// Never attach to, read output from, or collect data from the vendor browser.
const crypto=require('node:crypto');
const SCHEMA='local.fixture.worker-diagnostic.v1';
function byteBound(text,limit){const b=Buffer.from(text);if(b.length<=limit)return text;let end=limit;while(end>0&&(b[end]&0xc0)===0x80)end--;return b.subarray(0,end).toString('utf8');}
function safeText(value,fixtureBase){return String(value).replace(/(?:https?|wss?|data|blob):[^\s"'<>]+/gi,'[NON_FIXTURE_URL_REDACTED]').replace(/file:\/\/[^\s"'<>]+/g,url=>url.startsWith(fixtureBase)?url:'[NON_FIXTURE_FILE_REDACTED]');}
function classify(error){const m=String(error?.message||error);if(m==='coordination registry busy; retry once, then report stale lock (never steal it)')return 'REGISTRY_BUSY';if(/fixture event|bounded navigation events|owned fixture events/.test(m))return 'FIXTURE_EVENT_REFUSED';if(/ECC|check-write|admission refused|actor state/.test(m))return 'ECC_ADMISSION_OR_WRITE_REFUSED';if(/source|manifest|goal.*binding|input changed/.test(m))return 'SOURCE_OR_GOAL_BINDING_CHANGED';if(/Timeout|timeout|page\.|locator\.|route|viewport|navigation|validation|isolation|persistence/.test(m))return 'FIXTURE_ACTION_FAILURE';return 'UNCLASSIFIED_WORKER_FAILURE';}
function diagnostic(error,phase,fixtureBase){return {schema:SCHEMA,phase,classification:classify(error),message:byteBound(safeText(error?.message||error,fixtureBase),2048)};}
function createCollector(fixtureBase,limit=8192){const states=Object.fromEntries(['stdout','stderr'].map(k=>[k,{bytes:0,chunks:[],kept:0,hash:crypto.createHash('sha256')} ]));return {
 push(stream,value){const s=states[stream];if(!s)throw Error('exact child stream required');const b=Buffer.from(value);s.bytes+=b.length;s.hash.update(b);const n=Math.max(0,Math.min(b.length,limit-s.kept));if(n){s.chunks.push(b.subarray(0,n));s.kept+=n;}return states.stdout.bytes+states.stderr.bytes;},
 finish(){const out={};for(const [stream,s] of Object.entries(states)){const sanitized=safeText(Buffer.concat(s.chunks).toString('utf8'),fixtureBase),text=byteBound(sanitized,limit);out[stream]={bytes:s.bytes,capturedRawBytes:s.kept,retainedBytes:Buffer.byteLength(text),truncated:s.bytes>s.kept||Buffer.byteLength(sanitized)>limit,sha256:s.hash.digest('hex'),text};}out.structured=[];for(const stream of ['stdout','stderr'])for(const line of out[stream].text.split('\n')){try{const d=JSON.parse(line);if(d.schema===SCHEMA&&typeof d.phase==='string'&&typeof d.classification==='string'&&typeof d.message==='string')out.structured.push({schema:SCHEMA,phase:d.phase.slice(0,80),classification:d.classification.slice(0,80),message:byteBound(d.message,2048)});}catch{}}return out;}
 };}
function eventRecorder({base,clock,events,limit=128}){let enabled=false,failure=null;return {enable(){enabled=true;},disable(){enabled=false;},failure(){return failure;},record(kind,url){if(!enabled||failure)return;try{if(events.length>=limit)throw Error('bounded navigation events');const u=new URL(url);if(u.protocol!=='file:'||!u.href.startsWith(base))throw Error('only owned fixture events');events.push({kind,atNs:clock(),fixture:require('node:path').basename(u.pathname),fragment:u.hash==='#mobile-section'?'mobile-section':null});}catch(e){failure=diagnostic(e,'navigation-event',base);}},assertClean(){if(failure){const e=Error('fixture event refused: '+failure.message);e.fixtureDiagnostic=failure;throw e;}}};}
module.exports={SCHEMA,safeText,classify,diagnostic,createCollector,eventRecorder};
